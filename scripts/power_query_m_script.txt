// (Update the file path to where you saved blinkit_sales_raw.csv)
// ============================================================================

let
    Source = Csv.Document(
        File.Contents("C:\Path\To\blinkit_sales_raw.csv"),
        [Delimiter=",", Columns=12, Encoding=65001, QuoteStyle=QuoteStyle.None]
    ),
    PromotedHeaders = Table.PromoteHeaders(Source, [PromoteAllScalars=true]),
    ChangedTypes = Table.TransformColumnTypes(PromotedHeaders, {
        {"Item_Fat_Content", type text},
        {"Item_Identifier", type text},
        {"Item_Type", type text},
        {"Outlet_Establishment_Year", Int64.Type},
        {"Outlet_Identifier", type text},
        {"Outlet_Location_Type", type text},
        {"Outlet_Size", type text},
        {"Outlet_Type", type text},
        {"Item_Visibility", type number},
        {"Item_Weight", type number},
        {"Sales", type number},
        {"Rating", type number}
    }),
    // Standardize inconsistent Item_Fat_Content values
    ReplaceLF = Table.ReplaceValue(ChangedTypes, "LF", "Low Fat", Replacer.ReplaceText, {"Item_Fat_Content"}),
    ReplaceLowFat = Table.ReplaceValue(ReplaceLF, "low fat", "Low Fat", Replacer.ReplaceText, {"Item_Fat_Content"}),
    ReplaceReg = Table.ReplaceValue(ReplaceLowFat, "reg", "Regular", Replacer.ReplaceText, {"Item_Fat_Content"}),
    // Replace null Item_Weight with the column average
    AvgWeight = List.Average(ReplaceReg[Item_Weight]),
    FillNullWeight = Table.ReplaceValue(ReplaceReg, null, Number.Round(AvgWeight, 2), Replacer.ReplaceValue, {"Item_Weight"}),
    // Remove any duplicate rows
    RemovedDuplicates = Table.Distinct(FillNullWeight)
in
    RemovedDuplicates
